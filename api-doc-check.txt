<script>
    window.onload = function() {
        // Build a system
        const ui = SwaggerUIBundle({
            dom_id: '#swagger-ui',
            url: "https://astrodev.test8.io.vn/docs?api-docs.json",
            operationsSorter: null,
            configUrl: null,
            validatorUrl: null,
            oauth2RedirectUrl: "https://astrodev.test8.io.vn/api/oauth2-callback",

            requestInterceptor: function(request) {
                if (request.url.indexOf('/api/v1/') !== -1) {
                    request.headers['Accept'] = 'application/json';
                    if (request.method === 'POST' || request.method === 'PUT' || request.method === 'PATCH') {
                        if (!request.headers['Content-Type']) {
                            request.headers['Content-Type'] = 'application/json';
                        }
                    }
                    request.headers['X-CSRF-TOKEN'] = '';
                    const token = localStorage.getItem('auth_token');
                    if (token && !request.headers['Authorization']) {
                        request.headers['Authorization'] = 'Bearer ' + token;
                    }
                }
                return request;
            },

            responseInterceptor: function(response) {
                if (response.url.indexOf('/api/v1/auth/login') !== -1 && response.status === 200) {
                    try {
                        const body = typeof response.body === 'string' ? JSON.parse(response.body) : response.body;
                        const token = body?.data?.access_token || body?.access_token || body?.data?.token || body?.token;
                        if (token) {
                            localStorage.setItem('auth_token', token);
                        }
                    } catch (e) {
                        console.error('Error parsing token from login response', e);
                    }
                } else if (response.url.indexOf('/api/v1/auth/logout') !== -1 && response.status === 200) {
                    localStorage.removeItem('auth_token');
                }
                return response;
            },

            presets: [
                SwaggerUIBundle.presets.apis,
                SwaggerUIStandalonePreset
            ],

            plugins: [
                SwaggerUIBundle.plugins.DownloadUrl
            ],

            layout: "StandaloneLayout",
            docExpansion : "none",
            deepLinking: true,
            filter: true,
            persistAuthorization: "false",

        })

        window.ui = ui

            }
</script>
