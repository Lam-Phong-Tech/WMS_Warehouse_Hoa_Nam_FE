// @vitest-environment node
import fs from "node:fs";
import path from "node:path";
import ts from "typescript";
import { describe, expect, it } from "vitest";
const root=path.resolve("src/modules/reports");
function collect(dir:string):string[] { return fs.readdirSync(dir,{withFileTypes:true}).flatMap(e=>e.isDirectory()?collect(path.join(dir,e.name)):[path.join(dir,e.name)]); }
const files=collect(root).filter(p=>/\.tsx?$/.test(p)&&!/(?:\.(?:test|spec)\.|__tests__|fixtures)/.test(p));
describe("all report runtime sources",()=>{
  it("has no executable sample datasets or forced-preview branches",()=>{
    const problems:string[]=[];
    for(const file of files){
      const source=ts.createSourceFile(file,fs.readFileSync(file,"utf8"),ts.ScriptTarget.Latest,true,file.endsWith(".tsx")?ts.ScriptKind.TSX:ts.ScriptKind.TS);
      const visit=(node:ts.Node)=>{
        if(ts.isIdentifier(node)&&/^(?:makeDemoRows|DEMO_ROWS|DEMO_FROM|DEMO_TO|demoMode|forcedState|readPreviewState)$/.test(node.text))problems.push(path.relative(root,file)+":"+node.text);
        if(ts.isStringLiteral(node)&&/^2026-09-(04|10)$/.test(node.text))problems.push(path.relative(root,file)+":fixed sample period");
        if(ts.isImportDeclaration(node)&&ts.isStringLiteral(node.moduleSpecifier)&&/(?:fixtures|mock-data|demo-data)/i.test(node.moduleSpecifier.text))problems.push(path.relative(root,file)+":runtime fixture import");
        ts.forEachChild(node,visit);
      };visit(source);
    }
    expect(problems).toEqual([]);
  });
  it("retains the API adapters of every published report",()=>{
    const expected:Record<string,string>={
      "inventory-flow/InboundFlowReport.tsx":"useInventoryFlowSnapshot",
      "inventory-flow/OutboundFlowReport.tsx":"useInventoryFlowSnapshot",
      "stock-reconcile/StockReconcileReport.tsx":"useStockReconcileSnapshot",
      "movement-trace/MovementTraceReport.tsx":"useMovementTraceReport",
      "import-match/ImportQualityBaseline.tsx":"useSkuImportReports",
      "packing-label/PackingLabelPrintReport.tsx":"usePackingLabelReport",
      "warranty-components/WarrantyComponentsReport.tsx":"useWarrantyReport",
      "recipient-agency/RecipientAgencyReport.tsx":"useRecipient",
      "io-history/rpt08/Rpt08HistoryReport.tsx":"useIoHistorySnapshot",
      "io-history/IoHistoryReport.tsx":"useIoHistorySnapshot",
    };
    for(const [file,adapter] of Object.entries(expected))expect(fs.readFileSync(path.join(root,file),"utf8"),file).toContain(adapter);
  });
});
