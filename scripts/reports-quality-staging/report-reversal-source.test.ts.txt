import { afterEach, describe, expect, it, vi } from "vitest";
vi.mock("@/shared/api/http",()=>({http:{getPage:vi.fn()}}));
import { http } from "@/shared/api/http";
import { fetchReversalReportSource } from "./report-reversal-source";
const getPage=vi.mocked(http.getPage);
afterEach(()=>vi.resetAllMocks());
describe("complete reversal source for reports",()=>{
  it("combines every API page in order and preserves the requested period",async()=>{
    getPage.mockImplementation(async(_url,config)=>{const page=Number(config?.params?.page);return {data:[{id:String(page)}],meta:{last_page:3,current_page:page,per_page:1,total:3}} as never;});
    const result=await fetchReversalReportSource("2026-09-17","2026-09-23");
    expect(result.data.map(row=>row.id)).toEqual(["1","2","3"]);
    expect(getPage).toHaveBeenCalledTimes(3);
    for(const [,config] of getPage.mock.calls)expect(config?.params).toMatchObject({from_date:"2026-09-17",to_date:"2026-09-23",per_page:200});
  });
  it("does not publish the first page if a later API page fails",async()=>{
    getPage.mockResolvedValueOnce({data:[{id:"1"}],meta:{last_page:2,total:2}} as never).mockRejectedValueOnce(new Error("API page failed"));
    await expect(fetchReversalReportSource("from","to")).rejects.toThrow("API page failed");
  });
  it("rejects excessive paging rather than silently truncating totals",async()=>{
    getPage.mockResolvedValueOnce({data:[],meta:{last_page:51,total:10001}} as never);
    await expect(fetchReversalReportSource("from","to")).rejects.toThrow("incomplete");
    expect(getPage).toHaveBeenCalledTimes(1);
  });
  it("rejects a missing portion of the total reported by the API",async()=>{
    getPage.mockResolvedValueOnce({data:[{id:"1"}],meta:{last_page:1,total:2}} as never);
    await expect(fetchReversalReportSource("from","to")).rejects.toThrow("incomplete");
  });
  it("accepts an explicitly empty API dataset without creating records",async()=>{
    getPage.mockResolvedValueOnce({data:[],meta:{last_page:1,total:0}} as never);
    expect((await fetchReversalReportSource("from","to")).data).toEqual([]);
  });
});
